def load_iq(path):
    raw = np.fromfile(path, dtype='<f4')
    return raw[0::2] + 1j * raw[1::2]

def compare(expected, recovered):
    n = min(len(expected), len(recovered))
    matches = int(np.sum(expected[:n] == recovered[:n])) if n else 0
    accuracy = matches / n if n else 0.0
    return accuracy, 1.0 - accuracy, len(recovered) - len(expected), matches, n

def best_shift(expected, recovered):
    results = []
    for shift in range(-20, 21):
        if shift >= 0:
            left, right = expected[shift:], recovered[:len(expected) - shift]
        else:
            left, right = expected[:len(expected) + shift], recovered[-shift:]
        n = min(len(left), len(right))
        acc = float(np.mean(left[:n] == right[:n])) if n else 0.0
        results.append((acc, shift, n))
    return max(results)
